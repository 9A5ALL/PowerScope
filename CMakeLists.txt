cmake_minimum_required(VERSION 3.16)
project(PowerScope VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTKMM REQUIRED IMPORTED_TARGET gtkmm-3.0)
add_executable(powerscope main.cpp)
target_compile_definitions(powerscope PRIVATE POWERSCOPE_VERSION="${PROJECT_VERSION}")
target_link_libraries(powerscope PRIVATE PkgConfig::GTKMM)
target_compile_options(powerscope PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS powerscope RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/powerscope.png
    DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps
    RENAME org.powerscope.monitor.png
)
install(FILES packaging/org.powerscope.monitor.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
set(CPACK_GENERATOR "DEB")
set(CPACK_PACKAGE_NAME "powerscope")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_CONTACT "PowerScope contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native Linux battery and energy monitoring application")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
include(CPack)
